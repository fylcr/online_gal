cmake_minimum_required(VERSION 3.19)
project(artemis_compat_clean_web C CXX)
if(NOT EMSCRIPTEN)
    message(FATAL_ERROR "Clean Web baseline requires Emscripten")
endif()
set(manifest_path "${CMAKE_CURRENT_SOURCE_DIR}/source-manifest.json")
if(DEFINED ARTC_CLEAN_CLOCK AND NOT ARTC_CLEAN_CLOCK STREQUAL "")
    if(NOT ARTC_CLEAN_CLOCK STREQUAL "clock-ms-01")
        message(FATAL_ERROR "Only ARTC_CLEAN_CLOCK=clock-ms-01 is supported")
    endif()
    set(manifest_path "${CMAKE_CURRENT_SOURCE_DIR}/candidates/clock-ms-01/manifest.json")
endif()
file(READ "${manifest_path}" manifest)
string(JSON platform_count LENGTH "${manifest}" platformFiles)
math(EXPR platform_last "${platform_count} - 1")
foreach(i RANGE 0 ${platform_last})
    string(JSON platform_file GET "${manifest}" platformFiles ${i})
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${platform_file}")
        message(FATAL_ERROR "Lane B platform missing: ${platform_file}")
    endif()
endforeach()
file(TO_CMAKE_PATH "${ARTC_CLEAN_SOURCE}" ARTC_CLEAN_SOURCE)
if(NOT EXISTS "${ARTC_CLEAN_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "ARTC_CLEAN_SOURCE must name the recipe's guarded archive directory")
endif()
foreach(option ARTC_BUILD_CLI ARTC_BUILD_JNI ARTC_BUILD_MAC_HOST ARTC_BUILD_TESTS ARTC_TEST_GLES ARTC_ENABLE_FFMPEG)
    set(${option} OFF CACHE BOOL "Disabled for isolated Web baseline" FORCE)
endforeach()
add_library(artc_emscripten_gles INTERFACE)
set(ARTC_GLES_TARGET artc_emscripten_gles CACHE STRING "Emscripten GLES" FORCE)
set(ARTC_GRAPHICS_BACKEND GLES CACHE STRING "Web graphics" FORCE)
set(ARTC_AUDIO_BACKEND NULL CACHE STRING "Platform Audio::Impl supplied below" FORCE)
add_subdirectory("${ARTC_CLEAN_SOURCE}" upstream)
# Retain upstream sources except the manifest's complete TU replacements.
get_target_property(core_sources artemis_core SOURCES)
string(JSON replacement_count LENGTH "${manifest}" replacements)
math(EXPR replacement_last "${replacement_count} - 1")
foreach(i RANGE 0 ${replacement_last})
    string(JSON original MEMBER "${manifest}" replacements ${i})
    string(JSON replacement GET "${manifest}" replacements "${original}")
    set(original_path "${ARTC_CLEAN_SOURCE}/${original}")
    list(FIND core_sources "${original_path}" found)
    if(found EQUAL -1)
        message(FATAL_ERROR "Upstream source missing from target: ${original}")
    endif()
    if(ARTC_CLEAN_CLOCK STREQUAL "clock-ms-01" AND original STREQUAL "src/script/lua_engine.cpp")
        string(JSON expected_original GET "${manifest}" clockCandidate upstreamSha256)
        string(JSON expected_replacement GET "${manifest}" clockCandidate replacementSha256)
        file(SHA256 "${original_path}" actual_original)
        file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${replacement}" actual_replacement)
        if(NOT actual_original STREQUAL expected_original OR NOT actual_replacement STREQUAL expected_replacement)
            message(FATAL_ERROR "Clock candidate source hash mismatch")
        endif()
    endif()
    list(REMOVE_ITEM core_sources "${original_path}")
    list(APPEND core_sources "${CMAKE_CURRENT_SOURCE_DIR}/${replacement}")
endforeach()
set_property(TARGET artemis_core PROPERTY SOURCES "${core_sources}")
target_include_directories(artemis_core BEFORE PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(lua51 PRIVATE -fwasm-exceptions)
target_compile_options(artemis_core PRIVATE -fwasm-exceptions)
add_executable(artemis-compat-clean host/web_main.cpp)
target_include_directories(artemis-compat-clean PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(artemis-compat-clean PRIVATE cxx_std_17)
target_compile_options(artemis-compat-clean PRIVATE -Wall -Wextra -fwasm-exceptions)
target_link_libraries(artemis-compat-clean PRIVATE artemis::core)
string(JSON abi_count LENGTH "${manifest}" cAbi)
math(EXPR abi_last "${abi_count} - 1")
string(JSON allocator_count LENGTH "${manifest}" allocatorExports)
math(EXPR allocator_last "${allocator_count} - 1")
set(exports "")
foreach(i RANGE 0 ${allocator_last})
    string(JSON symbol GET "${manifest}" allocatorExports ${i})
    list(APPEND exports "${symbol}")
endforeach()
foreach(i RANGE 0 ${abi_last})
    string(JSON symbol GET "${manifest}" cAbi ${i} name)
    list(APPEND exports "_${symbol}")
endforeach()
list(JOIN exports "," export_names)
string(JSON method_count LENGTH "${manifest}" runtimeMethods)
math(EXPR method_last "${method_count} - 1")
set(methods "")
foreach(i RANGE 0 ${method_last})
    string(JSON method GET "${manifest}" runtimeMethods ${i})
    list(APPEND methods "${method}")
endforeach()
list(JOIN methods "," method_names)
# Worker owns timing and the SAB range bridge; these flags introduce no frame loop.
# Font rendering stays in upstream's vendored implementation; no new font port.
target_link_options(artemis-compat-clean PRIVATE
    --no-entry -fwasm-exceptions
    "-sENVIRONMENT=worker" "-sINVOKE_RUN=0" "-sASSERTIONS=1"
    "-sINITIAL_MEMORY=67108864" "-sALLOW_MEMORY_GROWTH=1"
    "-sMAXIMUM_MEMORY=4294967296" "-sSTACK_SIZE=4194304"
    "-sUSE_WEBGL2=1" "-sFULL_ES2=1" "-sFORCE_FILESYSTEM=1"
    "-sEXPORTED_FUNCTIONS=${export_names}" "-sEXPORTED_RUNTIME_METHODS=${method_names}"
    "-sEXPORT_ES6=0" "-sMODULARIZE=0")
set_target_properties(artemis-compat-clean PROPERTIES OUTPUT_NAME "artemis-compat" SUFFIX ".js")
